<script>
(function(){
  var el=document.getElementById("rb-count"); if(!el) return;
  var target=Date.UTC(2026,8,1,15,0,0);
  var q=function(k){return el.querySelector("[data-c='"+k+"']")};
  var D=q("d"),H=q("h"),M=q("m"),S=q("s");
  function pad(n){return n<10?"0"+n:""+n}
  function tick(){
    var left=target-Date.now(); if(left<0) left=0;
    var t=Math.floor(left/1000);
    D.textContent=Math.floor(t/86400);
    H.textContent=pad(Math.floor(t%86400/3600));
    M.textContent=pad(Math.floor(t%3600/60));
    S.textContent=pad(t%60);
  }
  tick(); setInterval(tick,1000);
})();
</script>
